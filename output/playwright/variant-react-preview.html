type ButtonProps = {
  size?: "small" | "medium" | "large";
  variant?: "primary" | "secondary";
  state?: "default" | "hover" | "disabled";
  icon?: "none" | "leading" | "trailing";
  children: React.ReactNode;
};

export function Button({
  size = "large",
  variant = "primary",
  state = "default",
  icon = "leading",
  children,
}: ButtonProps) {
  return <ButtonRoot $size={size} $variant={variant}
    disabled={state === "disabled"}>{children}</ButtonRoot>;
}
